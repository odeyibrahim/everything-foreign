/**
 * Get products from a category (cached)
 */
function get_products_from_category($category_slug) {
    $cache_key = 'yeno_products_' . $category_slug;
    $products_data = get_transient($cache_key);
    if ( $products_data !== false ) {
        return $products_data;
    }
    $products_data = array();
    $category = get_term_by('slug', $category_slug, 'product_cat');
    $args = array(
        'post_type'      => 'product',
        'posts_per_page' => -1,
        'tax_query'      => $category ? array(array('taxonomy' => 'product_cat', 'field' => 'slug', 'terms' => $category_slug)) : array(),
        'orderby'        => 'date',
        'order'          => 'DESC'
    );
    if (!$category) {
        $args['posts_per_page'] = 20;
        unset($args['tax_query']);
    }
    $products_query = new WP_Query($args);
    if ($products_query->have_posts()) {
        while ($products_query->have_posts()) {
            $products_query->the_post();
            $product_id = get_the_ID();
            $product = wc_get_product($product_id);
            if (!$product) continue;
            $gallery_ids = $product->get_gallery_image_ids();
            $gallery_urls = array();
            foreach ($gallery_ids as $gallery_id) {
                $gallery_urls[] = wp_get_attachment_url($gallery_id);
            }
            $image_url = wp_get_attachment_url($product->get_image_id());
            if (!$image_url) $image_url = wc_placeholder_img_src('medium');
            // Use wc_price for multi-currency
            if ($product->is_type('variable')) {
                $price_min = $product->get_variation_price('min');
                $price_max = $product->get_variation_price('max');
                $price_str = wc_price($price_min) . ' – ' . wc_price($price_max);
            } else {
                $price_str = wc_price($product->get_price());
            }
            $products_data[] = array(
                'id'          => $product_id,
                'name'        => $product->get_name(),
                'price'       => $product->get_price(),
                'priceRange'  => $price_str,
                'image'       => $image_url,
                'link'        => get_permalink($product_id),
                'variations'  => $gallery_urls,
                'featured'    => $product->is_featured(),
            );
        }
        wp_reset_postdata();
    }
    set_transient($cache_key, $products_data, 12 * HOUR_IN_SECONDS);
    return $products_data;
}

function renderProducts() {
    var grid = document.getElementById('productGrid');
    if (!grid) return;
    var products = window.azureProducts || [];
    if (products.length === 0) {
        grid.innerHTML = '<div style="grid-column:1/-1;text-align:center;padding:var(--space-2xl) 0;opacity:0.5;">No products found.</div>';
        return;
    }
    var html = '';
    products.forEach(function(p) {
        var priceStr = p.priceRange || '₦' + Number(p.price).toLocaleString();
        var hoverImages = '';
        if (p.variations && p.variations.length > 0) {
            var cycleImages = p.variations.slice(0, 4);
            hoverImages = cycleImages.map(function(v, idx) {
                var delay = idx * 1.2;
                return '<img class="variation-hover" src="'+v+'" alt="'+p.name+' variation" loading="lazy" style="transition-delay: '+delay+'s; opacity:0;">';
            }).join('');
        }
        var colorDots = '';
        if (p.variations && p.variations.length > 0) {
            var dots = p.variations.slice(0, 3);
            colorDots = '<div class="color-options">' +
                dots.map(function(v, idx) {
                    return '<span class="color-dot ' + (idx === 0 ? 'active' : '') + '" style="background-image:url('+v+');background-size:cover;background-position:center;" data-product="'+p.id+'" data-image="'+v+'"></span>';
                }).join('') +
                '</div>';
        }
        var featuredBadge = p.featured ? '<span style="position:absolute;top:0.5rem;left:0.5rem;background:var(--accent);color:var(--text-primary);font-size:0.5rem;padding:0.15rem 0.5rem;letter-spacing:0.06em;text-transform:uppercase;z-index:2;">Featured</span>' : '';
        html += '<div class="product-card" id="card-'+p.id+'">' +
            '<a href="'+p.link+'"><div class="image-wrap" style="position:relative;"><img src="'+p.image+'" alt="'+p.name+'" loading="lazy" id="img-'+p.id+'">' + hoverImages + featuredBadge + '</div></a>' +
            '<div class="info"><a href="'+p.link+'" class="title">'+p.name+'</a><span class="price">'+priceStr+'</span>' + colorDots +
            '<button class="btn btn-primary" style="margin-top:8px;width:100%;" onclick="addToWooCart('+p.id+')">Add to Bag</button></div></div>';
    });
    grid.innerHTML = html;

    // Color dot click handlers
    document.querySelectorAll('.color-dot').forEach(function(dot) {
        dot.addEventListener('click', function(e) {
            e.preventDefault();
            e.stopPropagation();
            var productId = this.dataset.product;
            var imageUrl = this.dataset.image;
            var img = document.getElementById('img-'+productId);
            if (img) img.src = imageUrl;
            var card = document.getElementById('card-'+productId);
            if (card) {
                card.querySelectorAll('.color-dot').forEach(function(d) { d.classList.remove('active'); });
                this.classList.add('active');
            }
        });
    });
}
