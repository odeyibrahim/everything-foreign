/**

if (!defined('ABSPATH')) exit;

function yeno_opt($key, $fallback = '') {
    if (function_exists('get_field')) {
        $val = get_field($key, 'option');
        if ($val !== null && $val !== false && $val !== '') return $val;
    }
    return $fallback;
}

function yeno_get_logo_url() {
    $logo_id = get_theme_mod('custom_logo');
    return $logo_id ? wp_get_attachment_image_url($logo_id, 'full') : '';
}

function yenolabel_get_cart_items_html() {
    if (!class_exists('WooCommerce') || !WC()->cart) {
        return '<div style="text-align:center;padding:40px 0;color:var(--text-tertiary);font-size:0.8rem;">' . __('Your bag is empty.', 'yenolabel') . '</div>';
    }
    ob_start();
    if (WC()->cart->is_empty()) {
        echo '<div style="text-align:center;padding:40px 0;color:var(--text-tertiary);font-size:0.8rem;">' . __('Your bag is empty.', 'yenolabel') . '</div>';
    } else {
        foreach (WC()->cart->get_cart() as $key => $item) {
            $product = $item['data'];
            if (!$product) continue;
            $variation = array();
            if (isset($item['variation'])) {
                foreach ($item['variation'] as $attr => $val) {
                    $label = wc_attribute_label(str_replace('attribute_', '', $attr));
                    $variation[] = $label . ': ' . ucfirst($val);
                }
            }
            echo '<div class="mini-cart__item" data-cart-key="' . esc_attr($key) . '">';
            echo '<img src="' . (get_the_post_thumbnail_url($product->get_id(), 'thumbnail') ?: wc_placeholder_img_src('thumbnail')) . '" class="mini-cart__item-img" loading="lazy">';
            echo '<div style="flex:1;">';
            echo '<div class="mini-cart__item-name">' . $product->get_name() . '</div>';
            echo '<div class="mini-cart__item-variant">' . implode(' / ', $variation) . '</div>';
            echo '<div class="mini-cart__item-price">' . wc_price($product->get_price() * $item['quantity']) . '</div>';
            echo '<div class="mini-cart__item-qty">';
            echo '<button class="qty-minus" data-key="' . esc_attr($key) . '">-</button>';
            echo '<span class="qty-count">' . $item['quantity'] . '</span>';
            echo '<button class="qty-plus" data-key="' . esc_attr($key) . '">+</button>';
            echo '</div>';
            echo '<button class="btn-remove" data-key="' . esc_attr($key) . '">' . __('Remove', 'yenolabel') . '</button>';
            echo '</div></div>';
        }
    }
    return ob_get_clean();
}

function yeno_wishlist_toggle() {
    check_ajax_referer('yenolabel_nonce', 'nonce');
    $product_id = intval($_POST['product_id']);
    $user_id = get_current_user_id();
    if (!$user_id) {
        $wishlist = isset($_COOKIE['yeno_wishlist']) ? json_decode(stripslashes($_COOKIE['yeno_wishlist']), true) : array();
        if (in_array($product_id, $wishlist)) {
            $wishlist = array_diff($wishlist, array($product_id));
            $action = 'removed';
        } else {
            $wishlist[] = $product_id;
            $action = 'added';
        }
        setcookie('yeno_wishlist', json_encode($wishlist), time() + 30 * DAY_IN_SECONDS, COOKIEPATH, COOKIE_DOMAIN);
        wp_send_json(['success' => true, 'action' => $action, 'count' => count($wishlist)]);
        return;
    }
    $wishlist = get_user_meta($user_id, 'yeno_wishlist', true) ?: array();
    if (in_array($product_id, $wishlist)) {
        $wishlist = array_diff($wishlist, array($product_id));
        $action = 'removed';
    } else {
        $wishlist[] = $product_id;
        $action = 'added';
    }
    update_user_meta($user_id, 'yeno_wishlist', $wishlist);
    wp_send_json(['success' => true, 'action' => $action, 'count' => count($wishlist)]);
}

            // Update count
            var count = 0;
            $('.mini-cart__item').each(function() { count++; });
            $('#cart-count, #mini-cart-count').text(count);
        });
    }

    $(document).on('submit', 'form.cart', function(e) {
        e.preventDefault();
        var form = $(this);
        var data = form.serialize() + '&add-to-cart=' + form.find('[name="add-to-cart"]').val();
        $.post(yenolabel_ajax.ajax_url, data, function(response) {
            if (response && response.success) {
                window.openCart();
                updateCartUI();
                window.showToast(yenolabel_i18n.added_to_cart, 'success');
            } else {
                window.showToast(yenolabel_i18n.error, 'error');
            }
        });

        var newQty = $(this).hasClass('qty-plus') ? currentQty + 1 : currentQty - 1;
        if (newQty < 1) {
            $.post(yenolabel_ajax.ajax_url, { action: 'remove_cart_item', key: key, nonce: yenolabel_ajax.nonce }, function() { updateCartUI(); });

        }

    window.openSearch = function() {
        $('#searchOverlay').addClass('open');
        $('#searchInput').focus();
    };
    window.closeSearch = function() {
        $('#searchOverlay').removeClass('open');
        $('#searchResults').html('');
    };

    window.performSearch = function() {
        var q = $('#searchInput').val().trim();
        clearTimeout(searchTimeout);
        if (q.length < 2) { $('#searchResults').html(''); return; }
        searchTimeout = setTimeout(function() {
            $.get(yenolabel_ajax.ajax_url, {
                action: 'yeno_search',
                q: q,
                nonce: yenolabel_ajax.nonce
            }, function(data) {
                var html = '';
                if (data.length === 0) html = '<div style="padding:20px;text-align:center;color:var(--text-tertiary);">No results</div>';
                else {
                    data.forEach(function(item) {
                        html += '<div class="search-result-item" onclick="window.location=\'' + item.url + '\'">';
                        html += '<img src="' + item.img + '" loading="lazy">';
                        html += '<div class="search-result__details"><div class="search-result__name">' + item.name + '</div><div class="search-result__price">' + item.price + '</div></div></div>';
                    });
                }
                $('#searchResults').html(html);
            });
        }, 300);
    };

    // Filter
    window.applyFilters = function() {
        var filters = { category: '', price_min: 0, price_max: 200000 };
        $('.filter-option.active').each(function() {
            var filter = $(this).data('filter');
            var value = $(this).data('value');
            if (filter === 'price_min') filters.price_min = parseInt(value);
            if (filter === 'price_max') filters.price_max = parseInt(value);
        });
        var currentCat = $('.filter-scroll .btn-underline.active');
        if (currentCat.length) filters.category = currentCat.text().trim().toLowerCase();

        $.post(yenolabel_ajax.ajax_url, {
            action: 'yeno_filter_products',
            filters: JSON.stringify(filters),
            nonce: yenolabel_ajax.nonce
        }, function(response) {
            if (response.success) {
                var grid = $('#productGrid');
                if (response.count === 0) {
                    grid.html('<p style="grid-column:1/-1;text-align:center;color:var(--text-tertiary);padding:var(--space-2xl) 0;">No products.</p>');
                    return;
                }
                var html = '';
                response.products.forEach(function(p) {
                    html += '<div class="product-card" onclick="window.location=\'' + p.link + '\'">' +
                        '<div class="product-card__img-wrap"><img src="' + p.image + '" class="product-card__img" loading="lazy" alt="' + p.name + '"></div>' +
                        '<div class="product-card__title">' + p.name + '</div>' +
                        '<div class="product-card__price">' + p.price + '</div></div>';
                });
                grid.html(html);
            }
        });
    };
    window.clearFilters = function() {
        $('.filter-option.active').removeClass('active');
        window.applyFilters();
    };
    $(document).on('click', '.filter-option', function() { $(this).toggleClass('active'); });

    // Frontpage product render
    function renderProducts() {
        var grid = document.getElementById('productGrid');
        if (!grid) return;
        var products = window.azureProducts || [];
        if (products.length === 0) {
            grid.innerHTML = '<div style="grid-column:1/-1;text-align:center;padding:var(--space-2xl) 0;opacity:0.5;">No products.</div>';
            return;
        }
        var html = '';
        products.forEach(function(p) {
            var priceStr = p.priceRange || '₦' + Number(p.price).toLocaleString();
            var colorDots = '';
            if (p.variations && p.variations.length > 0) {
                var dots = p.variations.slice(0, 3);
                colorDots = '<div class="color-options">' +
                    dots.map(function(v, idx) {
                        return '<span class="color-dot ' + (idx === 0 ? 'active' : '') + '" style="background-image:url(' + v + ');background-size:cover;background-position:center;" data-product="' + p.id + '" data-image="' + v + '" role="button"></span>';
                    }).join('') +
                    '</div>';
            }
            html += '<div class="product-card" id="card-' + p.id + '">' +
                '<a href="' + p.link + '"><div class="image-wrap"><img src="' + p.image + '" alt="' + p.name + '" loading="lazy" id="img-' + p.id + '"></div></a>' +
                '<div class="info"><a href="' + p.link + '" class="title">' + p.name + '</a><span class="price">' + priceStr + '</span>' + colorDots +
                '<button class="btn btn-primary" onclick="addToWooCart(' + p.id + ')">Add to Bag</button></div></div>';
        });
        grid.innerHTML = html;

        document.querySelectorAll('.color-dot').forEach(function(dot) {
            dot.addEventListener('click', function(e) {
                e.preventDefault();
                e.stopPropagation();
                var productId = this.dataset.product;
                var imageUrl = this.dataset.image;
                var img = document.getElementById('img-' + productId);
                if (img) img.src = imageUrl;
                var card = document.getElementById('card-' + productId);
                if (card) {
                    card.querySelectorAll('.color-dot').forEach(function(d) { d.classList.remove('active'); });
                    this.classList.add('active');
                }
            });
        });
    }

    window.addToWooCart = function(productId) {
        $.post(yenolabel_ajax.ajax_url, {
            action: 'add_to_cart',
            product_id: productId,
            quantity: 1,
            nonce: yenolabel_ajax.nonce
        }, function(response) {
            if (response.success) {
                window.openCart();
                window.showToast(yenolabel_i18n.added_to_cart, 'success');
            } else {
                window.showToast(yenolabel_i18n.error, 'error');
            }
        });
    };

    $(document).ready(function() {
        if (document.getElementById('productGrid')) renderProducts();
    });
})(jQuery);
